Automatic state persistence with versioning

🎯 Feature Composition Example

Here's how multiple features work together in the cart store:

export const CartStore = signalStore(
  { providedIn: 'root' },
  
  // Core state
  withState<CartState>({ /* initial state */ }),
  withEntities<CartItem>(),
  
  // Custom features - reusable and composable!
  withTaxCalculator(),
  withDiscounts(),
  withPersistence({
    key: 'cart-store',
    storage: 'localStorage',
    version: 1,
    exclude: ['isLoading', 'error']
  }),
  withAnalytics({ enablePerformanceTracking: true }),
  
  // Enhanced computed properties using features
  withComputed((store) => ({
    enhancedCartSummary: computed(() => {
      const items = store.entities();
      const taxResult = store.calculateTax(/* ... */);
      const discountResult = store.calculateDiscounts(/* ... */);
      return { /* enhanced summary */ };
    })
  })),
  
  // Methods that leverage features
  withMethods((store) => ({
    addItem(product: Product) {
      // Add to cart
      patchState(store, addEntity(newItem));
      
      // Track with analytics
      store.trackCartAdd(product.id, product.price);
      
      // Auto-apply best discounts
      store.autoApplyBestDiscounts(/* ... */);
    }
  }))
);
